```typescript
const REGIONES_CHILE = [

  'Valparaíso', 'Metropolitana', "O'Higgins", 'Maule', 'Ñuble', 'Biobío',
  'Araucanía', 'Los Ríos', 'Los Lagos', 'Aysén', 'Magallanes',
]

function openCreate() {
  setEditingPref(null)
  setFormName('')
  setFormKeywords('')
  setFormMinAmount('')
  setFormMaxAmount('')
  setFormRegiones([])
  setFormNotifyEmail(true)
  setFormNotifyInApp(true)
  setFormNotifyWebhook(false)
  setPrefError(null)
  setShowModal(true)
}

function openEdit(p: AlertPreference) {
  setEditingPref(p)
  setFormName(p.name)
  setFormKeywords((p.keywords ?? []).join(', '))
  setFormMinAmount(p.min_amount != null ? String(p.min_amount) : '')
  setFormMaxAmount(p.max_amount != null ? String(p.max_amount) : '')
  setFormRegiones(p.regiones ?? [])
  setFormNotifyEmail(p.notify_email)
  setFormNotifyInApp(p.notify_in_app)
  setFormNotifyWebhook(p.notify_webhook)
  setPrefError(null)
  setShowModal(true)
}

async function handleSavePref() {
  if (!formName.trim()) { setPrefError('El nombre es obligatorio'); return }
  setSaving(true); setPrefError(null)
  const payload: Record<string, unknown> = {
    name: formName.trim(),
    keywords: formKeywords.split(',').map((k) => k.trim()).filter(Boolean),
    min_amount: formMinAmount ? Number(formMinAmount) : null,
    max_amount: formMaxAmount ? Number(formMaxAmount) : null,
    regiones: formRegiones.length > 0 ? formRegiones : null,
    notify_email: formNotifyEmail,
    notify_in_app: formNotifyInApp,
    notify_webhook: formNotifyWebhook,
  }
  try {
    if (editingPref) {
      await apiClient.updateAlertPreference(editingPref.id, payload)
    } else {
      await apiClient.createAlertPreference(payload)
    }
    setShowModal(false)
    loadPreferences()
  } catch (err: any) {
    setPrefError(err?.response?.data?.detail ?? 'Error al guardar preferencia')
  } finally {
    setSaving(false)
  }
}

async function handleDeletePref(p: AlertPreference) {
  if (!confirm(`¿Eliminar la preferencia "${p.name}"?`)) return
  try {
    await apiClient.deleteAlertPreference(p.id)
    loadPreferences()
  } catch (err: any) {
    alert(err?.response?.data?.detail ?? 'Error al eliminar')
  }
}
```
